import os
import json
from datetime import datetime
from typing import Literal
from dotenv import load_dotenv
from google import genai
from pydantic import BaseModel, Field

# Carrega variáveis de ambiente do arquivo .env
load_dotenv()


# 1. Definição do Schema Estruturado (Garante que a IA responda sempre no formato exato)
class TicketTriagem(BaseModel):
    nome_cliente: str = Field(description="Nome do remetente ou empresa, se identificado. Caso contrário, 'Não informado'.")
    categoria: Literal["Suporte Técnico", "Comercial / Orçamento", "Financeiro", "Outros"] = Field(
        description="Categoria principal da solicitação."
    )
    urgencia: Literal["Alta", "Média", "Baixa"] = Field(
        description="Nível de urgência baseado no impacto relatado (ex: sistema fora do ar = Alta)."
    )
    resumo_acao: str = Field(description="Resumo objetivo em 1 frase do que precisa ser feito.")
    setor_destino: str = Field(description="Fila ou equipe interna que deve receber o chamado.")


# 2. Função Principal do Agente de Triagem
def processar_mensagem_com_ia(mensagem_bruta: str) -> TicketTriagem:
    client = genai.Client(api_key=os.environ.get("GEMINI_API_KEY"))

    prompt = f"""
    Você é um agente de triagem e automação operacional.
    Analise a mensagem recebida abaixo, extraia as informações principais,
    classifique a urgência com critério técnico e defina o setor responsável.

    Mensagem recebida:
    \"\"\"{mensagem_bruta}\"\"\"
    """

    response = client.models.generate_content(
        model="gemini-2.5-flash",
        contents=prompt,
        config={
            "response_mime_type": "application/json",
            "response_schema": TicketTriagem,
            "temperature": 0.1,  # Baixa temperatura para respostas consistentes e determinísticas
        },
    )

    # Valida e converte o JSON retornado pela IA no objeto Pydantic
    dados_validados = TicketTriagem.model_validate_json(response.text)
    return dados_validados


# 3. Execução Automática (Simula disparo para Webhook / Sistema Interno / Log)
def executar_roteamento(ticket: TicketTriagem) -> None:
    registro = {
        "timestamp": datetime.now().isoformat(),
        "status": "ROTEADO_AUTOMATICAMENTE",
        "dados": ticket.model_dump(),
    }

    print("\n--- RESULTADO DA TRIAGEM AUTOMÁTICA ---")
    print(json.dumps(registro, indent=2, ensure_ascii=False))

    if ticket.urgencia == "Alta":
        print(f"🚨 [ALERTA CRÍTICO] Disparando notificação imediata para: {ticket.setor_destino}")
    else:
        print(f"✅ [FILA PADRÃO] Chamado registrado para: {ticket.setor_destino}")


# 4. Demonstração Prática com Casos Reais
if __name__ == "__main__":
    exemplos_entrada = [
        "Oi, aqui é o Carlos da Transportadora Alfa. Nosso sistema de emissão de notas travou agora e temos 15 caminhões parados no pátio esperando liberação!",
        "Bom dia! Gostaria de saber quanto fica o plano mensal para 10 usuários e se vocês emitem boleto faturado.",
        "Olá, paguei a fatura ontem mas ainda consta como em aberto no portal. Podem verificar?",
    ]

    for i, msg in enumerate(exemplos_entrada, start=1):
        print(f"\nProcessando entrada #{i}...")
        resultado = processar_mensagem_com_ia(msg)
        executar_roteamento(resultado)
